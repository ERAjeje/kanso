  <action>
    In `backend/cmd/kanso-api/main.go`, add the following function before `main()`:

    ```go
    func securityHeaders(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            w.Header().Set("Content-Security-Policy",
                "default-src 'self'; script-src 'self' https://accounts.google.com; frame-src https://accounts.google.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; connect-src 'self' https://fcm.googleapis.com; img-src 'self' data:")
            w.Header().Set("Strict-Transport-Security", "max-age=31536000; includeSubDomains")
            w.Header().Set("X-Content-Type-Options", "nosniff")
            w.Header().Set("X-Frame-Options", "DENY")
            w.Header().Set("Referrer-Policy", "strict-origin-when-cross-origin")
            w.Header().Set("Permissions-Policy", "geolocation=(), microphone=(), camera=()")
            next.ServeHTTP(w, r)
        })
    }
    ```

    Then add it to the middleware chain after CORS (line 120, before Timeout):

    ```go
    r.Use(cors.Handler(cors.Options{...}))
    r.Use(securityHeaders)
    r.Use(chimiddleware.Timeout(30 * time.Second))
    ```
