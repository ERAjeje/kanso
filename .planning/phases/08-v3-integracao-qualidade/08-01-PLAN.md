        url := fmt.Sprintf("%s/kanso-config/config:active_users", c.baseURL)
        req, err := http.NewRequest("GET", url, nil)
        if err != nil {
            return nil, fmt.Errorf("new request: %w", err)
        }
        req.SetBasicAuth(c.adminUser, c.adminPass)

        resp, err := c.httpClient.Do(req)
        if err != nil {
            return nil, fmt.Errorf("get active users: %w", err)
        }
        defer resp.Body.Close()

        if resp.StatusCode == http.StatusNotFound {
            return nil, nil
        }
        if resp.StatusCode >= 300 {
            return nil, fmt.Errorf("get active users status: %d", resp.StatusCode)
        }

        var doc ActiveUsersDoc
        if err := json.NewDecoder(resp.Body).Decode(&doc); err != nil {
            return nil, fmt.Errorf("decode active users: %w", err)
        }
        return &doc, nil
    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Google OAuth -> kanso-api | Google-verified id_token, trust on first use |
| kanso-api -> CouchDB | Admin credentials, full access |
| User -> Login endpoint | Public HTTP endpoint, rate-limited by Google |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-01 | Spoofing | Google OAuth token | mitigate | Google verifies id_token signature. Backend validates with Google certs before allowlist check. |
| T-08-02 | Elevation of Privilege | isEmailAllowed | mitigate | Document `config:active_users` protected by CouchDB admin auth. Only admin can modify. |
| T-08-03 | Denial of Service | GetActiveUsers | accept | Single CouchDB GET per login. If CouchDB is down, login fails (acceptable for personal app). |
| T-08-04 | Information Disclosure | Error message "email not authorized" | accept | Reveals email is not in allowlist but not which emails are. Acceptable for personal app. |
</threat_model>
