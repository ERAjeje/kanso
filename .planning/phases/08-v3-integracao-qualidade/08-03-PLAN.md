    1. In `backend/cmd/kanso-api/main.go`, add a new middleware function before `main()`:

    ```go
    func internalNetworkOnly(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            host := r.RemoteAddr
            // Extract IP before port
            ipStr := host
            if idx := strings.LastIndex(host, ":"); idx != -1 {
                ipStr = host[:idx]
            }
            ip := net.ParseIP(ipStr)
            if ip == nil {
                http.Error(w, `{"error":"forbidden"}`, http.StatusForbidden)
                return
            }
            // Allow Docker internal networks (172.16.0.0/12, 10.0.0.0/8) and localhost
            if ip.IsLoopback() ||
                (ip[0] == 10) ||
                (ip[0] == 172 && ip[1] >= 16 && ip[1] <= 31) {
                next.ServeHTTP(w, r)
                return
            }
            http.Error(w, `{"error":"forbidden"}`, http.StatusForbidden)
        })
    }
    ```

    Add imports `"net"` and `"strings"` to main.go.

    2. Wrap the `/api/push/send` route with the middleware:

    ```go
    r.With(internalNetworkOnly).Post("/api/push/send", pushHandler.HandleSend)
    ```

    ```go
    r.Group(func(r chi.Router) {
        r.Use(internalNetworkOnly)
        r.Post("/api/push/send", pushHandler.HandleSend)
    })
    ```
