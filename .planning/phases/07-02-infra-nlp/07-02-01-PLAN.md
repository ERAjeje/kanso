    ```python
    def compute_pos_weights(label_matrix: np.ndarray) -> torch.Tensor:
        num_examples = label_matrix.shape[0]
        pos_counts = label_matrix.sum(axis=0)
        neg_counts = num_examples - pos_counts
        pos_weights = neg_counts / np.maximum(pos_counts, 1)
        pos_weights = np.clip(pos_weights, 0.5, 50.0)  # clip to prevent extreme weights
        return torch.FloatTensor(pos_weights)
    ```

    ```python
    import pytest
    import numpy as np
    from src.model_config import LABELS, NUM_LABELS, THRESHOLD, MODEL_VERSION

    def test_goemotions_labels_count():
        assert len(GOEMOTIONS_LABELS) == 28
        assert NUM_GOEMOTIONS == 28
        assert NUM_OUR_LABELS == 13


    def test_label_map_coverage():
        """Every GoEmotions label maps to at least one of our labels."""
        for ge_label in GOEMOTIONS_LABELS:
            assert ge_label in LABEL_MAP, f"Missing mapping for {ge_label}"
            mapped_indices = LABEL_MAP[ge_label]
            assert len(mapped_indices) >= 1
            assert len(mapped_indices) <= 2
            for idx in mapped_indices:
                assert 0 <= idx < 13, f"Index {idx} out of range for {ge_label}"


    def test_map_goemotions_row_single_label():
        """Single GoEmotion label maps correctly."""
        # admiration -> amor (10), gratidão (11)
        binary_28 = [0] * 28
        binary_28[GOEMOTIONS_LABELS.index("admiration")] = 1
        result = map_goemotions_row(binary_28)
        assert len(result) == 13
        assert result[10] == 1  # amor
        assert result[11] == 1  # gratidão


    def test_map_goemotions_row_empty():
        """No labels -> all zeros."""
        result = map_goemotions_row([0] * 28)
        assert result == [0] * 13


    def test_map_goemotions_row_neutral():
        binary_28 = [0] * 28
        binary_28[GOEMOTIONS_LABELS.index("neutral")] = 1
        result = map_goemotions_row(binary_28)
        assert result[12] == 1  # neutro


    def test_compute_pos_weights_shape_and_clip():
        """compute_pos_weights returns tensor of correct shape with clipped values."""
        # Simulate: 100 examples, label 0 appears in 80%, label 5 appears in 2%
        label_matrix = np.zeros((100, 13))
        label_matrix[:80, 0] = 1  # alegria is common
        label_matrix[:2, 5] = 1   # surpresa is rare
        weights = compute_pos_weights(label_matrix)
    ```
