    ```python
    import logging
    import subprocess
    import torch
    from transformers import AutoTokenizer, AutoModelForSequenceClassification
    from src.model_config import LABELS, NUM_LABELS, THRESHOLD, MODEL_PATH, MAX_LENGTH, MODEL_VERSION

    logger = logging.getLogger(__name__)


    def get_model_version() -> str:
        """Return git commit hash, or fallback to configured MODEL_VERSION (D-34)."""
        try:
            result = subprocess.run(
                ["git", "rev-parse", "--short", "HEAD"],
                capture_output=True, text=True,
                timeout=5,
            )
            if result.returncode == 0 and result.stdout.strip():
                return result.stdout.strip()
        except (FileNotFoundError, subprocess.TimeoutExpired):
            pass
        return MODEL_VERSION


    class EmotionClassifier:
        """CPU-based emotion classifier using fine-tuned BERTimbau (D-33).

        Loads model once at construction time. predict() uses torch.no_grad()
        for efficient single-request inference. No batching (D-32).
        """

        def __init__(self, model_path: str = MODEL_PATH):
            logger.info("Loading emotion classifier from %s", model_path)
            self.device = torch.device("cpu")
            self.tokenizer = AutoTokenizer.from_pretrained(model_path)
            self.model = AutoModelForSequenceClassification.from_pretrained(model_path)
            self.model.to(self.device)
            self.model.eval()
            logger.info("EmotionClassifier loaded (%d labels)", NUM_LABELS)

        @torch.no_grad()
        def predict(self, text: str) -> dict:
            """Run inference on a single text input.

            Args:
                text: Portuguese text to analyze (sensações + contexto + pensamentos)

            Returns:
                dict with keys:
                    - emotion_principal: str (label with highest score, D-26)
                    - emotions: list[dict] (all labels above THRESHOLD, D-25)
                    - scores: dict[str, float] (all label scores 0.0-1.0)
                    - intensidade: float (max score, D-26)
                    - modelo_versao: str (D-34)
            """
            inputs = self.tokenizer(
                text,
                return_tensors="pt",
                truncation=True,
                padding=True,
                max_length=MAX_LENGTH,
            )
            inputs = {k: v.to(self.device) for k, v in inputs.items()}
            outputs = self.model(**inputs)
            probabilities = torch.sigmoid(outputs.logits).squeeze()

            # Handle single-label edge case (squeeze removes batch dim)
            if probabilities.dim() == 0:
                probabilities = probabilities.unsqueeze(0)

            scores_list = probabilities.tolist()

            # Build scores dict (all labels)
            scores_dict = {
                self.labels[i]: round(float(scores_list[i]), 4)
                for i in range(NUM_LABELS)
            }

            # Find labels above threshold (D-25)
            above_threshold = [
                (LABELS[i], float(scores_list[i]))
                for i in range(NUM_LABELS)
                if scores_list[i] >= THRESHOLD
            ]

            # D-27: If nothing above threshold, return neutro with its score
            if not above_threshold:
                neutro_idx = LABELS.index("neutro")
                above_threshold = [("neutro", float(scores_list[neutro_idx]))]

            # D-26: emotion_principal = label with max score
            above_threshold.sort(key=lambda x: x[1], reverse=True)
            emotion_principal = above_threshold[0][0]
            intensidade = max(scores_list)

            return {
                "emotion_principal": emotion_principal,
                "emotions": [
                    {"emotion": e, "score": round(s, 4)}
                    for e, s in above_threshold
                ],
                "scores": scores_dict,
                "intensidade": round(float(intensidade), 4),
            }

        @property
        def labels(self) -> list:
            return LABELS
    ```

    ```python
    import torch
    from src.model_config import LABELS, THRESHOLD

    WEIGHTED_F1_THRESHOLD = 0.70  # D-28: minimum acceptable Weighted F1

    # Each entry: (portuguese_text, expected_primary_emotion)
    TEST_PHRASES = [
        # alegria (alegria)
        ("Estou muito feliz hoje", "alegria"),
        ("Que dia maravilhoso", "alegria"),
        ("Consegui passar no exame", "alegria"),
        ("Adorei a surpresa que você me fez", "alegria"),
        ("Estou radiante com a notícia", "alegria"),
        # ... ~45 more alegria phrases
        # Total: ~50 per label × 13 labels
    ]

    def test_phrases_count():
        """D-30: ~50 phrases per emotion, 13 emotions = ~650 total."""
        from collections import Counter
        labels = [p[1] for p in TEST_PHRASES]
        counts = Counter(labels)
        for label in LABELS:
            assert counts[label] >= 40, f"{label} only has {counts[label]} phrases"
        assert len(TEST_PHRASES) >= 13 * 40  # at least 520 total

    def test_phrases_labels_valid():
        """All expected labels must be in the 13-label set."""
        for _, expected_label in TEST_PHRASES:
            assert expected_label in LABELS, f"Invalid label: {expected_label}"

    @pytest.mark.skipif(
        not os.path.exists(os.environ.get("MODEL_PATH", "/model")),
        reason="No trained model artifact at MODEL_PATH — run training first"
    )
    def test_weighted_f1_threshold():
        """D-28: Weighted F1 >= 0.70 on curated test set.

        Loads the fine-tuned BERTimbau model from MODEL_PATH, runs inference
        on all TEST_PHRASES, computes Weighted F1 as the primary evaluation
        metric, and asserts it meets the WEIGHTED_F1_THRESHOLD.
        """
        import evaluate
        from src.classifier import EmotionClassifier
        from src.model_config import MODEL_PATH

        model_path = os.environ.get("MODEL_PATH", MODEL_PATH)
        classifier = EmotionClassifier(model_path)

        y_true = []
        y_pred = []

        for text, expected_label in TEST_PHRASES:
            result = classifier.predict(text)
            # Build multi-label vectors: true label is one-hot for expected_label,
            # predicted vector comes from thresholded model output
            true_vec = [1 if label == expected_label else 0 for label in LABELS]
            pred_vec = [
                1 if score >= THRESHOLD else 0
                for label, score in result["scores"].items()

            # If nothing above threshold, emotion_principal wins
            if sum(pred_vec) == 0:
                pred_vec[LABELS.index(result["emotion_principal"])] = 1
            y_true.append(true_vec)
            y_pred.append(pred_vec)

        f1_metric = evaluate.load("f1", config_name="multilabel")
        result = f1_metric.compute(
            predictions=y_pred, references=y_true, average="weighted"
        )
        weighted_f1 = result["f1"]

        print(f"\nWeighted F1: {weighted_f1:.4f} (threshold: {WEIGHTED_F1_THRESHOLD})")
        assert weighted_f1 >= WEIGHTED_F1_THRESHOLD, (
            f"Weighted F1 {weighted_f1:.4f} < {WEIGHTED_F1_THRESHOLD} — "
            f"run D-29 iteration (more data + hyperparameter tuning)"
        )
    ```

    ```python
    import pytest
    import torch
    from src.classifier import EmotionClassifier, get_model_version
    from src.model_config import LABELS, THRESHOLD, NUM_LABELS


    class TestEmotionClassifierContract:
        """Contract tests for EmotionClassifier interface.
        
        These tests validate the inference structure assuming a model exists at MODEL_PATH.
        They use a tiny BERTimbau model for testing, or can be skipped if no model available.
        """

        def test_predict_structure(self, classifier):
            """predict() returns dict with all required keys."""
            result = classifier.predict("Estou muito feliz hoje")
            assert isinstance(result, dict)
            assert "emotion_principal" in result
            assert "emotions" in result
            assert "scores" in result
            assert "intensidade" in result

        def test_predict_emotion_principal_is_string(self, classifier):
            result = classifier.predict("Estou com medo")
            assert isinstance(result["emotion_principal"], str)
            assert result["emotion_principal"] in LABELS

        def test_predict_emotions_is_list(self, classifier):
            result = classifier.predict("Que dia lindo")
            assert isinstance(result["emotions"], list)
            assert len(result["emotions"]) >= 1
            for e in result["emotions"]:
                assert "emotion" in e
                assert "score" in e
                assert e["emotion"] in LABELS
                assert 0 <= e["score"] <= 1

        def test_predict_scores_all_labels(self, classifier):
            result = classifier.predict("Estou cansado")
            assert isinstance(result["scores"], dict)
            assert len(result["scores"]) == NUM_LABELS
            for label in LABELS:
                assert label in result["scores"]
                assert 0 <= result["scores"][label] <= 1

        def test_predict_intensidade_is_max_score(self, classifier):
            """D-26: intensidade = max emotion score."""
            result = classifier.predict("Estou muito bem")
            max_score = max(result["scores"].values())
            assert abs(result["intensidade"] - max_score) < 0.001

        def test_predict_empty_text_returns_neutro(self, classifier):
            """D-27: Short/uninformative input -> neutro."""
            result = classifier.predict("")
            assert result["emotion_principal"] == "neutro"

        def test_predict_short_text_returns_neutro(self, classifier):
            """D-27: Very short text should also lean neutro."""
            result = classifier.predict("ok")
    ```

    ```python
        logger.info("EmotionClassifier loaded from %s", _model_path)
    except Exception as e:
        logger.warning(
            "EmotionClassifier not available at %s (%s). "
            "Will lazy-init on first Analyze() call.",
            _model_path, e,
        )


    class AnalysisServicer(analysis_pb2_grpc.AnalysisServiceServicer):

        async def Analyze(self, request, context):
            logger.info("Analyze called for registro %s", request.registro_id)
            # Combine registration text fields for analysis (D-10: sensações + contexto + pensamentos)
            text_parts = []
            if request.sensacoes:
                text_parts.append(request.sensacoes)
            if request.contexto:
                text_parts.append(request.contexto)
            if request.pensamentos:
                text_parts.append(request.pensamentos)
            text = " ".join(text_parts)

            if not text.strip():
                # D-27: Short/uninformative input → neutro with high confidence
                return analysis_pb2.AnalyzeResponse(
                    emotion_principal="neutro",
                    emotions=[analysis_pb2.EmotionScore(emotion="neutro", score=0.95)],
                    scores={"neutro": 0.95},
                    intensidade=0.95,
                    analise_adicional="Texto vazio — classificado como neutro",
                    modelo_versao=_model_version
                )

            # Run inference — synchronous in thread pool to avoid blocking event loop
            loop = asyncio.get_event_loop()
            result = await loop.run_in_executor(None, self._predict_sync, text)

            return analysis_pb2.AnalyzeResponse(
                emotion_principal=result["emotion_principal"],
                emotions=[
                    analysis_pb2.EmotionScore(emotion=e["emotion"], score=e["score"])
                    for e in result["emotions"]
                ],
                scores=result["scores"],
                intensidade=result["intensidade"],
                analise_adicional="",
                modelo_versao=_model_version,
            )

        def _predict_sync(self, text: str) -> dict:
            """Synchronous inference wrapper (runs in thread pool).
            
            If _classifier is None (module-level init failed because no model
            artifact exists), attempts lazy initialization. Returns a neutral
            fallback if model is still unavailable.
            """
            global _classifier
            if _classifier is None:
                try:
                    _classifier = get_classifier(_model_path)
                except Exception as e:
                    logger.error("Failed to lazy-init classifier: %s", e)
                    return {
                        "emotion_principal": "neutro",
                        "emotions": [{"emotion": "neutro", "score": 1.0}],
                        "scores": {"neutro": 1.0},
                        "intensidade": 1.0,
                    }
            return _classifier.predict(text)


    async def serve_grpc(port: int = 50051):
        server = grpc.aio.server(futures.ThreadPoolExecutor(max_workers=10))
        analysis_pb2_grpc.add_AnalysisServiceServicer_to_server(AnalysisServicer(), server)
        server.add_insecure_port(f"[::]:{port}")
        logger.info("gRPC server starting on port %d", port)
        await server.start()
        await server.wait_for_termination()


    if __name__ == "__main__":
        asyncio.run(serve_grpc())
    ```
